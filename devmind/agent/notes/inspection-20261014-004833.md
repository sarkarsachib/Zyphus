"""
The agent's body.

The engine already carries biology-inspired modules; this file is the wiring
that turns them into one body state the agent can feel, report and be seen in:

- `src/heartbeat_engine.py`    the pulse: real rhythm types, beats and coherence
                               over a short burst, at a rate set by actual load.
- `src/emotion_200d_system.py` the mood: a 200-dimension emotional state, set
  `mood_stabilizer.py`         from what the agent has actually been doing and
  `cluster_modulation.py`      then stabilised into its emotion clusters.
                               (these three need numpy - see requirements.txt)

Everything here is optional by design: a module that cannot be imported is
reported as unavailable with its reason, and the rest of the body still works.
`vitals()` is the single reading the agent's `vitals` tool and the dashboard share.
"""

import asyncio
import os
import time
from typing import Any, Dict, List, Optional

REPO_ROOT = os.path.dirname(os.path.dirname(os.path.dirname(os.path.abspath(__file__))))

# Which of the engine's rhythms the body starts from, chosen by real load. The
# engine then adapts on its own from the energy and stress it is given.
RHYTHM_BY_LOAD = [(0.85, "GAMMA"), (0.62, "BETA"), (0.38, "ALPHA"), (0.16, "THETA")]
IDLE_RHYTHM = "DELTA"


# --- load, the body's sense of how hard it is working ------------------------


def _load() -> float:
    """0.0 (idle) .. 1.0 (flat out), from the container's own CPU and memory."""
    try:
        import psutil

        cpu = psutil.cpu_percent(interval=0.15) / 100.0
        memory = psutil.virtual_memory().percent / 100.0
        return round(min(1.0, max(0.0, cpu * 0.7 + memory * 0.3)), 3)
    except Exception:
        try:
            load1 = os.getloadavg()[0]
            cpus = os.cpu_count() or 1
            return round(min(1.0, max(0.0, load1 / cpus)), 3)
        except Exception:
            return 0.2


# --- the pulse ---------------------------------------------------------------


def pulse(intensity: float, stress: float = 0.0) -> Dict[str, Any]:
    """Run the engine's real heartbeat for a short burst and read the result.

    The engine models brain rhythms, not a human pulse: each of its patterns is a
    neural band (gamma 40 Hz, beta 20 Hz, alpha 10 Hz, theta 6 Hz, delta 2 Hz), and
    it adapts the rhythm itself from the energy and stress it is given. So the
    reading is a rhythm and a rate in Hz, never a heart rate in bpm.
    """
    try:
        from src.heartbeat_engine import HeartbeatEngine, HeartbeatType
    except Exception as exc:
        return {"available": False, "error": f"heartbeat engine: {type(exc).__name__}: {exc}"}

    rhythm_name = next((name for threshold, name in RHYTHM_BY_LOAD if intensity >= threshold),
                       IDLE_RHYTHM)
    engine = HeartbeatEngine()
    rhythm = HeartbeatType[rhythm_name]
    target_bpm = 60000.0 / engine.rhythm_patterns[rhythm].base_interval_ms

    async def burst():
        started = await engine.start(rhythm, target_bpm)
        if not started.get("success"):
            return started, {}, {}
        await engine.set_energy_level(intensity)
        await engine.set_stress_level(stress)
        await asyncio.sleep(0.5)
        status, metrics = engine.get_status(), engine.get_metrics()
        await engine.stop()
        return started, status, metrics

    try:
        started, status, metrics = asyncio.run(burst())
    except Exception as exc:
        return {"available": False, "error": f"heartbeat: {type(exc).__name__}: {exc}"}

    if not status:
        return {"available": False, "error": started.get("error", "heartbeat did not start")}
    settled = HeartbeatType[status["current_rhythm"].upper()]
    pattern = engine.rhythm_patterns[settled]
    return {
        "available": True,
        "rhythm": status["current_rhythm"],
        "rate_hz": round(1000.0 / pattern.base_interval_ms, 2),
        "rhythm_meaning": pattern.description,
        "started_from": rhythm_name.lower(),
        "beats_felt": status["total_beats"],
        "energy": round(status["energy_level"], 3),
        "stress": round(status["stress_level"], 3),
        "coherence": round(status["coherence_score"], 3),
        "stability": round(metrics.get("rhythm_stability", 0.0), 3),
    }


# --- the mood ----------------------------------------------------------------


def mood(activity: Optional[Dict[str, Any]] = None) -> Dict[str, Any]:
    """A real 200-dimension emotional state, driven by recent agent activity."""
    activity = activity or {}
    success = float(activity.get("success_rate", 0.7))
    backlog = float(activity.get("backlog", 0.0))
    try:
        from src.emotion_200d_system import EmotionDimension, EmotionState
        from src.mood_stabilizer import MoodStabilizer
    except Exception as exc:
        return {"available": False, "error": f"{type(exc).__name__}: {exc}"}

    try:
        state = EmotionState()
        state.update_emotion(EmotionDimension.JOY, 0.35 + 0.6 * success)
        state.update_emotion(EmotionDimension.TRUST, 0.5 + 0.3 * success)
        state.update_emotion(EmotionDimension.FEAR, 0.15 + 0.5 * (1.0 - success))
        state.update_emotion(EmotionDimension.ANXIETY, 0.1 + 0.6 * (1.0 - success))
        state.update_emotion(EmotionDimension.FRUSTRATION, 0.1 + 0.5 * backlog)
        state.update_emotion(EmotionDimension.CURIOSITY, 0.75)
        state.update_emotion(EmotionDimension.CALM, 0.7 - 0.4 * backlog)
        state = MoodStabilizer().stabilize_all_clusters(state)

        dims = state.dimension_values
        strongest = sorted(dims.items(), key=lambda item: -item[1])[:5]
        return {
            "available": True,
            "dimensions": len(dims),
            "label": strongest[0][0].value if strongest else "unknown",
            "strongest": [{"emotion": dim.value, "value": round(value, 3)} for dim, value in strongest],
            "stabilised_clusters": ["positive", "negative", "neutral"],
        }
    except Exception as exc:
        return {"available": False, "error": f"mood: {type(exc).__name__}: {exc}"}


# --- modules the body is built from -----------------------------------------


def links() -> List[Dict[str, str]]:
    """Which of the engine's own modules the body is actually wired to."""
    wanted = [
        ("heartbeat_engine", "src/heartbeat_engine.py"),
        ("emotion_200d_system", "src/emotion_200d_system.py"),
        ("mood_stabilizer", "src/mood_stabilizer.py"),
        ("cluster_modulation", "src/cluster_modulation.py"),
        ("interrupt_watchers", "src/interrupt_watchers.py"),
    ]
    status = []
    for name, rel in wanted:
        try:
            __import__(name if name.startswith("src.") else f"src.{name}")
            status.append({"module": rel, "status": "linked"})
        except Exception as exc:
            status.append({"module": rel, "status": f"unavailable ({type(exc).__name__})"})
    return status


def vitals(activity: Optional[Dict[str, Any]] = None) -> Dict[str, Any]:
    """One reading of the whole body: pulse, mood, and what it is wired to."""
    activity = activity or {}
    intensity = _load()
    stress = round(1.0 - float(activity.get("success_rate", 1.0)), 3)
    beat = pulse(intensity, stress)
    feeling = mood(activity)
    return {
        "at": time.strftime("%Y-%m-%d %H:%M:%S"),
        "load": intensity,
        "pulse": beat,
        "mood": feeling,
        "links": links(),
    }


def summary(reading: Dict[str, Any]) -> str:
    """A short spoken-style sentence about the body, used by the voice routine."""
    beat = reading.get("pulse", {})
    feeling = reading.get("mood", {})
    if not beat.get("available"):
        return (f"My rhythm is not available ({beat.get('error', 'unknown reason')}) "
                f"and I feel {feeling.get('label', 'unknown')}.")
    return (f"My mind is running at {beat['rate_hz']} Hz in {beat['rhythm']} rhythm - "
            f"{beat['rhythm_meaning'].lower()} - load {reading['load']}, "
            f"energy {beat['energy']}, and I feel {feeling.get('label', 'unknown')}.")
